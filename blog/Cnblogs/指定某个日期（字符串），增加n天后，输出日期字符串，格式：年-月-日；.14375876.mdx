 * [dateAddDays 从某个日期增加n天后的日期]
 * @param  {[string]} dateStr  [日期字符串]
 * @param  {[int]} dayCount [增加的天数]
 * @return {[string]}[增加n天后的日期字符串]
 </span><span style="color: #008000;">*/</span>
<span style="color: #0000ff;">function</span><span style="color: #000000;"> dateAddDays(dateStr,dayCount) {
    </span><span style="color: #0000ff;">var</span> tempDate=<span style="color: #0000ff;">new</span> Date(dateStr.replace(/-/g,"/"));<span style="color: #008000;">//</span><span style="color: #008000;">把日期字符串转换成日期格式</span>
    <span style="color: #0000ff;">var</span> resultDate=<span style="color: #0000ff;">new</span> Date((tempDate/1000+(86400*dayCount))*1000);//增加n天后的日期
    <span style="color: #0000ff;">var</span> resultDateStr=resultDate.getFullYear()+"-"+(resultDate.getMonth()+1)+"-"+(resultDate.getDate());<span style="color: #008000;">//</span><span style="color: #008000;">将日期转化为字符串格式</span>
    <span style="color: #0000ff;">return</span><span style="color: #000000;"> resultDateStr;
}
